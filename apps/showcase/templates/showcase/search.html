Every hit on this page is a title, a line of context and a link. When a model deserves more, give its CRUDView a transform_hit: whatever the dict returns beyond display and subtitle travels with the hit as extra — to this page, the omnibar, /api/search/ and the MCP tool alike. The blog does exactly this, so a published post renders as an article card instead of a row. 

# apps/blog/views.py
class PostCRUDView(CRUDView):
    enable_search  = True
    search_fields  = ["title", "excerpt", "body_markdown"]

    def get_search_variants(self):
        return {"default": "Article card", "compact": "Title + permalink"}

    def transform_hit(self, post, variant="default"):
        if variant == "compact":
            return {"display": post.title, "url": post.public_url}
        return {
            "display": post.title, "subtitle": post.excerpt,
            "card": "article",                    # the template switches on this
            "blog": post.blog.name, "published_at": post.published_at,
            "reading_minutes": words // 220,
            "tags": [t.name for t in post.tags.all()],
            "cover_url": post.cover_image.url,
        }
